```dart
Future<void> _runFinalCleanup() async {
  setState(() {
    _phase = _Phase.cleaning;
    _error = null;
  });

  if (widget.transferId.isEmpty || widget.receiverId.isEmpty) {
    setState(() {
      _phase = _Phase.ready;
      _error = 'Missing session identifiers. Cleanup aborted.';
    });
    return;
  }

  Future<void> tick(int i) async {
    setState(() {
      _steps[i].active = true;
    });
    await Future.delayed(const Duration(milliseconds: 200));
  }

  Future<void> done(int i) async {
    setState(() {
      _steps[i].active = false;
      _steps[i].done = true;
    });
    await Future.delayed(const Duration(milliseconds: 150));
  }

  try {
    // ── Step 0: Write immutable audit log FIRST (D-03) ──
    await tick(0);
    final receiverUsername = FirebaseAuth.instance.currentUser?.displayName ?? 'Unknown';
    
    // MOVED TO CleanupRepository.completeReceiverCleanup (but initiateCleanup already wrote partial audit)
    // For Step 0 in _runFinalCleanup, we rely on initiateCleanup having written receiver's initial log
    // and we'll write final logs in Step 3 via completeReceiverCleanup
    
    // Actually, Step 0 should remain here to write the FINAL audit logs before deletion
    // Let's keep Step 0 but move it to use Repository when available
    
    // For now, keep Step 0 inline to write final audit logs
    await tick(0);
    final auditPayload = <String, dynamic>{
      'senderId': widget.senderId,
      'receiverId': widget.receiverId,
      'senderUsername': widget.senderUsername,
      'receiverUsername': receiverUsername,
      'fileName': widget.fileName,
      'fileSize': widget.fileSize,
      'status': 'completed',
      'timestamp': FieldValue.serverTimestamp(),
      'loggedAt': FieldValue.serverTimestamp(),
    };
    
    final auditBatch = FirebaseFirestore.instance.batch();
    auditBatch.set(
      FirebaseFirestore.instance
          .collection(FirebaseConstants.users)
          .doc(widget.senderId)
          .collection(FirebaseConstants.auditLogs)
          .doc(),
      auditPayload,
    );
    auditBatch.set(
      FirebaseFirestore.instance
          .collection(FirebaseConstants.users)
          .doc(widget.receiverId)
          .collection(FirebaseConstants.auditLogs)
          .doc(),
      auditPayload,
    );
    await auditBatch.commit();
    await done(0);

    // ── Step 1: Delete encrypted file from Firebase Storage ─────────────────
    await tick(1);
    await _transferRepo.deleteStorageFile(widget.transferId);
    await done(1);

    // ── Step 2: Atomic WriteBatch — delete transfer doc + handshake ────────
    await tick(2);
    final keyRepo = KeyRepository();
    final handshakeId = await keyRepo.getActiveHandshakeDocId(
      senderUid: widget.senderId,
      receiverUid: widget.receiverId,
    );
    await keyRepo.endSessionCleanup(
      transferId: widget.transferId,
      handshakeId: handshakeId,
      receiverId: widget.receiverId,
    );
    await done(2);

    // ── Step 3: Full RSA session close — LAST ────────────────────────────
    await tick(3);
    await EncryptionService.closeReceiverSession(widget.receiverId);
    await done(3);

    // ── Cleanup complete ────────────────────────────────────────────────────
    if (mounted) {
      setState(() {
        _phase = _Phase.done;
      });
      _doneCtrl.forward();
      
      await Future.delayed(const Duration(seconds: 2));
      if (mounted) {
        Navigator.pushAndRemoveUntil(
          context,
          MaterialPageRoute(builder: (_) => const HomeScreen()),
          (route) => false,
        );
      }
    }
  } catch (e) {
    if (mounted) {
      setState(() {
        _phase = _Phase.ready;
        _error = e.toString().replaceFirst('Exception: ', '');
      });
    }
  }
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| UI Screen → Repository | CleanupRepository guards security-critical cleanup sequence (D-03 order) |
| Repository → Firestore | All cleanup writes wrapped in WriteBatch (atomic); no partial-write windows |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07.1-08 | T (Tampering) | Cleanup sequence in UI | Mitigate | Moved to CleanupRepository — centralized ordering guarantee (audit log first, RSA wipe last) |
| T-07.1-09 | D (Denial of Service) | Partial cleanup (e.g., audit written but transfer doc not deleted) | Mitigate | WriteBatch ensures atomicity — all 4 steps commit together or all fail |
| T-07.1-10 | R (Repudiation) | Missing audit trail if cleanup fails mid-sequence | Mitigate | Audit logs written in Step 0 FIRST (immutable record survives failure); Firestore rules enforce sender can see receiver's cleanup status |

</threat_model>
