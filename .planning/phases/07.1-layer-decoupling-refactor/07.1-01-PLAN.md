import 'dart:typed_data';
import 'package:cloud_firestore/cloud_firestore.dart';
import '../../core/constants/firebase_constants.dart';
import '../../core/errors/app_exceptions.dart';

class CleanupRepository {
  final FirebaseFirestore _db;

  CleanupRepository({
    FirebaseFirestore? db,
  }) : _db = db ?? FirebaseFirestore.instance;

  /// Turn 1: Receiver initiates cleanup by writing cleanup_initiated flag and audit log
  Future<void> initiateReceiverCleanup({
    required String transferId,
    required String receiverId,
    required String senderId,
    required String senderUsername,
    required String fileName,
    required int fileSize,
  }) async {
    try {
      final batch = _db.batch();
      
      // Write cleanup_initiated flag to transfer doc
      batch.update(
        _db.collection(FirebaseConstants.transfers).doc(transferId),
        {'cleanup_initiated': true},
      );
      
      // Write receiver's personal audit record
      batch.set(
        _db
            .collection(FirebaseConstants.users)
            .doc(receiverId)
            .collection(FirebaseConstants.auditLogs)
            .doc(),
        {
          'senderId': senderId,
          'receiverId': receiverId,
          'senderUsername': senderUsername,
          'fileName': fileName,
          'fileSize': fileSize,
          'status': 'completed',
          'timestamp': FieldValue.serverTimestamp(),
          'loggedAt': FieldValue.serverTimestamp(),
        },
      );
      
      await batch.commit();
    } on FirebaseException catch (e) {
      throw TransferException(
        'Cleanup initiation failed: ${e.message ?? e.code}',
        code: e.code,
      );
    }
  }

  /// Watch transfer doc for deletion (Turn 4 trigger)
  Stream<DocumentSnapshot> watchTransferForDeletion(String transferId) {
    return _db
        .collection(FirebaseConstants.transfers)
        .doc(transferId)
        .snapshots();
  }

  /// Turn 3: Complete full cleanup sequence with audit logs and batch deletes
  Future<void> completeReceiverCleanup({
    required String transferId,
    required String receiverId,
    required String senderId,
    required String senderUsername,
    required String fileName,
    required int fileSize,
  }) async {
    try {
      final batch = _db.batch();
      
      // Write final audit logs for both parties FIRST
      final auditPayload = {
        'senderId': senderId,
        'receiverId': receiverId,
        'senderUsername': senderUsername,
        'fileName': fileName,
        'fileSize': fileSize,
        'status': 'completed',
        'timestamp': FieldValue.serverTimestamp(),
        'loggedAt': FieldValue.serverTimestamp(),
      };
      
      // Sender's personal audit record
      batch.set(
        _db
            .collection(FirebaseConstants.users)
            .doc(senderId)
            .collection(FirebaseConstants.auditLogs)
            .doc(),
        auditPayload,
      );
      
      // Receiver's personal audit record
      batch.set(
        _db
            .collection(FirebaseConstants.users)
            .doc(receiverId)
            .collection(FirebaseConstants.auditLogs)
            .doc(),
        auditPayload,
      );
      
      // Delete transfer doc
      batch.delete(_db.collection(FirebaseConstants.transfers).doc(transferId));
      
      await batch.commit();
    } on FirebaseException catch (e) {
      throw TransferException(
        'Cleanup completion failed: ${e.message ?? e.code}',
        code: e.code,
      );
    }
  }
}

import 'package:cloud_firestore/cloud_firestore.dart';
import '../../core/constants/firebase_constants.dart';
import '../../core/errors/app_exceptions.dart';

class UserRepository {
  final FirebaseFirestore _db;

  UserRepository({
    FirebaseFirestore? db,
  }) : _db = db ?? FirebaseFirestore.instance;

  // ─── Update user document fields ───────────────────────────────────────────

  /// Update user document with provided fields (onboarding, preferences, etc.)
  Future<void> updateUser(
    String userId,
    Map<String, dynamic> fields,
  ) async {
    try {
      await _db
          .collection(FirebaseConstants.users)
          .doc(userId)
          .update(fields);
    } on FirebaseException catch (e) {
      throw AuthException(
        'Failed to update user: ${e.message ?? e.code}',
        code: e.code,
      );
    }
  }

  // ─── Stream user with friendships for connection map ─────────────────────

  /// Stream user document with their friendships (for connection node map visualization)
  Stream<Map<String, dynamic>?> streamUserWithFriendships(String userId) {
    return _db
        .collection(FirebaseConstants.users)
        .doc(userId)
        .snapshots()
        .asyncMap((userDoc) async {
          if (!userDoc.exists) return null;
          
          final userData = userDoc.data() ?? {};
          
          // Fetch user's friends
          final friendships = await _db
              .collection(FirebaseConstants.users)
              .doc(userId)
              .collection(FirebaseConstants.connections)
              .where('status', isEqualTo: 'accepted')
              .get();
          
          userData['friendships'] = friendships.docs
              .map((doc) => doc.data())
              .toList();
          
          return userData;
        });
  }
}

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Presentation Layer → Repository Layer | Untrusted input from UI screens (user IDs, transfer IDs) sanitized in Repository before Firestore queries |
| Repository Layer → Firestore | All queries/writes include security context (user ID, collection names) validated before execution |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07.1-01 | S (Spoofing) | Repository method parameters | Mitigate | Validate userId and transferId format (non-empty strings) before Firestore operations; rely on Firebase Auth for identity |
| T-07.1-02 | I (Information Disclosure) | Scattered Firestore calls in UI | Mitigate | Centralize all data access in Repositories → single audit point for sensitive queries |
| T-07.1-03 | T (Tampering) | WriteBatch operations in CleanupRepository | Mitigate | All batch operations atomic; no partial writes; errors caught and wrapped in TransferException |
| T-07.1-04 | R (Repudiation) | Audit log writes | Mitigate | Audit logs written FIRST in CleanupRepository.completeReceiverCleanup() before any deletion; immutable record |

</threat_model>
